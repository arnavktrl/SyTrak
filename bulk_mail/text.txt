const express = require("express");
const multer = require("multer");
const csv = require("csv-parser");
const fs = require("fs");
const path = require("path");
const sgMail = require("@sendgrid/mail");
require("dotenv").config();

const app = express();
const upload = multer({ dest: "uploads/" });
sgMail.setApiKey(process.env.SENDGRID_API_KEY);

// POST /send-csv-emails
app.post("/send-csv-emails", upload.single("recipients"), async (req, res) => {
  const results = [];

  fs.createReadStream(req.file.path)
    .pipe(csv())
    .on("data", (data) => results.push(data))
    .on("end", async () => {
      fs.unlinkSync(req.file.path); // remove uploaded CSV file

      const emailSubject = "A note from the Krishworks team";
      const emailBody = fs.readFileSync(
        path.join(__dirname, "index.html"),
        "utf8"
      );

      try {
        for (const row of results) {
          const toEmail = row.email?.trim();
          if (!toEmail) {
            console.warn("Missing or invalid email in row:", row);
            continue;
          }

          const msg = {
            to: toEmail,
            // cc: [process.env.CC_MAIL],
            from: {
              email: process.env.SENDER_EMAIL,
              name: process.env.SENDER_NAME,
            },
            replyTo: {
              email: process.env.SENDER_EMAIL,
              name: process.env.SENDER_NAME,
            },
            subject: emailSubject,
            html: emailBody.replace(
                    "{{name}}",
                    row.name
                  ),
            mailSettings: {
              subscription_tracking: {
                enable: true,
              },
            },
          };

          await sgMail.send(msg);
        }

        res.status(200).json({ message: "Bulk emails sent successfully" });
      } catch (error) {
        console.error("SendGrid error:", error.response?.body || error.message);
        res.status(500).json({ error: "Failed to send emails" });
      }
    });
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
